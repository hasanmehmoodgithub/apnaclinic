🔧 Extensions Guide

What are Extensions?

Extensions add new methods to existing classes without modifying them:

  • Context Extensions - Easy access to MediaQuery, screen size, etc.
  • Text Style Extensions - Responsive text sizing
  • DateTime Extensions - Date formatting and time ago

🚀 Quick Start

Import Extensions

// Import all extensions
import 'core/extensions/extensions_export.dart';

// Or import specific extension
import 'core/extensions/context_extensions.dart';
import 'core/extensions/text_style_extensions.dart';
import 'core/extensions/datetime_extensions.dart';

📱 Context Extensions

Screen Size

// Get screen dimensions
final width = context.screenWidth;
final height = context.screenHeight;
final size = context.screenSize;

// Check orientation
if (context.isLandscape) { }
if (context.isPortrait) { }

Responsive Sizing

// Get percentage of screen
final halfWidth = context.widthPercent(50);  // 50% of screen width
final quarterHeight = context.heightPercent(25);  // 25% of screen height

// Responsive sizing based on design
final responsiveWidth = context.responsiveWidth(100);  // Scales from design width
final responsiveHeight = context.responsiveHeight(50);  // Scales from design height

Screen Type

// Check screen size
if (context.isSmallScreen) { }   // width < 600
if (context.isMediumScreen) { } // 600 <= width < 1200
if (context.isLargeScreen) { }   // width >= 1200

// Check device type
if (context.isPhone) { }   // width < 600
if (context.isTablet) { } // width >= 600

Safe Areas

// Get safe area padding
final topPadding = context.statusBarHeight;
final bottomPadding = context.bottomSafeArea;
final padding = context.padding;

// Check keyboard
if (context.isKeyboardVisible) { }
final keyboardHeight = context.keyboardHeight;

Complete Example

import 'core/extensions/context_extensions.dart';

class MyWidget extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Container(
      width: context.widthPercent(80),  // 80% of screen width
      height: context.responsiveHeight(200),  // Responsive height
      padding: EdgeInsets.only(
        top: context.statusBarHeight,
        bottom: context.bottomSafeArea,
      ),
      child: Column(
        children: [
          if (context.isTablet)
            Text('Tablet View'),
          if (context.isPhone)
            Text('Phone View'),
        ],
      ),
    );
  }
}

📝 Text Style Extensions

Responsive Text

import 'core/extensions/text_style_extensions.dart';

// Method 1: Using extension on TextStyle
Text(
  'Hello',
  style: TextStyle(fontSize: 16).responsive(
    context: context,
    baseSize: 16,
    minSize: 12,
    maxSize: 20,
  ),
)

// Method 2: Using context extension
Text(
  'Hello',
  style: context.responsiveTextStyle(
    baseSize: 16,
    minSize: 12,
    maxSize: 20,
    color: Colors.black,
    fontWeight: FontWeight.bold,
  ),
)

// Method 3: Get responsive font size
Text(
  'Hello',
  style: TextStyle(
    fontSize: context.responsiveFontSize(16, minSize: 12, maxSize: 20),
  ),
)

Predefined Text Styles

// Small text
Text('Small', style: context.smallTextStyle)

// Medium text
Text('Medium', style: context.mediumTextStyle)

// Large text
Text('Large', style: context.largeTextStyle)

// Extra large text
Text('Extra Large', style: context.extraLargeTextStyle)

📅 DateTime Extensions

Date Formatting

import 'core/extensions/datetime_extensions.dart';

final date = DateTime.now();

// Formatted date
date.toFormattedDate();      // "Jan 15, 2024"
date.toShortDate();          // "15/01/2024"
date.toIsoDate();            // "2024-01-15"

// Time formatting
date.toTime();               // "14:30"
date.to12HourTime();         // "02:30 PM"

// Date and time
date.toFormattedDateTime();  // "Jan 15, 2024 14:30"
date.toShortDateTime();      // "15/01/2024 14:30"

Time Ago

final date = DateTime.now().subtract(Duration(hours: 2));

// Time ago
date.timeAgo();              // "2 hours ago"
date.timeAgo(numericDates: false);  // "Today"

// Relative time
date.relativeTime();         // "2 hours ago" or "in 2 hours"

Date Checks

final date = DateTime.now();

// Check date
date.isToday;      // true
date.isYesterday;  // false
date.isTomorrow;   // false
date.isPast;       // false
date.isFuture;     // false
date.isThisWeek;   // true
date.isThisMonth;  // true
date.isThisYear;   // true

Date Calculations

final date = DateTime.now();

// Get start/end
date.startOfDay;     // 00:00:00
date.endOfDay;       // 23:59:59
date.startOfWeek;    // Monday 00:00:00
date.endOfWeek;      // Sunday 23:59:59
date.startOfMonth;   // First day of month
date.endOfMonth;     // Last day of month

// Add/subtract
date.addDays(5);           // Add 5 days
date.subtractDays(3);       // Subtract 3 days
date.addMonths(2);         // Add 2 months
date.addYears(1);          // Add 1 year

// Differences
date.daysDifference(otherDate);
date.hoursDifference(otherDate);
date.minutesDifference(otherDate);

// Check if between
date.isBetween(startDate, endDate);

// Get age
date.age;  // Age in years

✅ Best Practices

1. Use Context Extensions for Responsive Design

✅ Good 
Container(
  width: context.widthPercent(80),
  padding: EdgeInsets.only(top: context.statusBarHeight),
)
❌ Bad - Hardcoded 
Container(
  width: 300,
  padding: EdgeInsets.only(top: 50),
)

2. Use Responsive Text for All Devices

✅ Good 
Text('Hello', style: context.responsiveTextStyle(baseSize: 16))
❌ Bad - Fixed size 
Text('Hello', style: TextStyle(fontSize: 16))

3. Use DateTime Extensions for Formatting

✅ Good 
Text(date.timeAgo())
Text(date.toFormattedDate())
❌ Bad - Manual formatting 
Text('${date.day}/${date.month}/${date.year}')
